import java.util.Scanner;

public class PageOptimal {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Input
        System.out.print("Enter number of frames: ");
        int frames = sc.nextInt();

        System.out.print("Enter number of pages: ");
        int pages = sc.nextInt();

        int[] ref = new int[pages];
        System.out.println("Enter the reference string:");
        for (int i = 0; i < pages; i++) {
            ref[i] = sc.nextInt();
        }

        int[] memory = new int[frames];
        for (int i = 0; i < frames; i++) {
            memory[i] = -1;
        }

        int hits = 0, faults = 0;

        for (int i = 0; i < pages; i++) {
            int page = ref[i];
            boolean hit = false;

            // Check if page already in memory
            for (int j = 0; j < frames; j++) {
                if (memory[j] == page) {
                    hits++;
                    hit = true;
                    break;
                }
            }

            // If not found -> FAULT
            if (!hit) {
                faults++;
                int replaceIndex = -1;

                // Look for empty frame first
                for (int j = 0; j < frames; j++) {
                    if (memory[j] == -1) {
                        replaceIndex = j;
                        break;
                    }
                }

                // If no empty frame, replace optimal page
                if (replaceIndex == -1) {
                    int farthest = -1;
                    int index = -1;

                    for (int j = 0; j < frames; j++) {
                        int nextUse = -1;

                        // Look ahead to find next use of memory[j]
                        for (int k = i + 1; k < pages; k++) {
                            if (ref[k] == memory[j]) {
                                nextUse = k;
                                break;
                            }
                        }

                        // If not used again, choose this immediately
                        if (nextUse == -1) {
                            index = j;
                            break;
                        }

                        // Else, pick page with farthest next use
                        if (nextUse > farthest) {
                            farthest = nextUse;
                            index = j;
                        }
                    }

                    replaceIndex = index;
                }

                // Replace page
                memory[replaceIndex] = page;
            }

            // Print current memory status
            System.out.print("Step " + (i + 1) + " (Page " + page + "): ");
            for (int j = 0; j < frames; j++) {
                if (memory[j] == -1)
                    System.out.print("- ");
                else
                    System.out.print(memory[j] + " ");
            }
            System.out.println(hit ? " (HIT)" : " (FAULT)");
        }

        System.out.println("\nTotal Page Faults = " + faults);
        System.out.println("Total Hits = " + hits);
    }
}
